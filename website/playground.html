import torch
from torch import nn
import seedl as sd

class ResidualMLP(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(4, 6)
        self.relu1 = nn.ReLU()
        self.res_fc = nn.Linear(6, 6)
        self.relu2 = nn.ReLU()
        self.fc_out = nn.Linear(6, 2)

    def forward(self, x):
        h1 = self.relu1(self.fc1(x))
        h2 = self.relu2(h1 + self.res_fc(h1))
        return self.fc_out(h2)

model = ResidualMLP().eval()
x = torch.tensor([[0.5, -1.2, 0.8, 0.3]])
view = sd.flow(model, x, mode="graph")
view.save("trace.html")

Gradient Health Dashboard

Diagnostic telemetry for training stability. Inspect parameter-level and layer-level L1 and L2 gradient norms, gradient-to-weight magnitude ratios, exact zero-gradient fractions, and nonfinite (NaN/Inf) occurrences. Zero-gradient fractions provide observational diagnostics rather than definitive diagnoses of dead neurons, as zero gradients can occur from sparse inputs, unused parameters, activation saturation, or disconnected computational paths. Helps identify vanishing gradients, exploding gradients, and numerical bottlenecks early.

Training · Playback

Training Playback

Sixteen real forward/backward passes through a 2 → 4 → 1 PyTorch network. Play, pause, rewind, choose a step, or inspect a neuron. The loss chart, weights, activations, and existing gradients come from each captured pass. Live playback means recorded snapshots, not continuous training streamed from a server.

Recorded by SeeDL 0.5.0 · All interaction stays in your browser · On narrow screens, scroll plots sideways
View the Python source
import torch
from torch import nn
import seedl as sd

torch.manual_seed(41)
model = nn.Sequential(nn.Linear(2, 4), nn.Tanh(), nn.Linear(4, 1))
x = torch.tensor([[1.0, -2.0]])
optimizer = torch.optim.SGD(model.parameters(), lr=0.08)
with sd.watch(model, name="Tiny regression", max_snapshots=16,
              max_bytes=512_000, max_values=12,
              max_neurons=12, max_connections=16) as run:
    for step in range(16):
        optimizer.zero_grad(set_to_none=True)
        output = model(x)
        loss = (output - 0.5).square().mean()
        loss.backward()
        run.capture(step=step, epoch=step // 4, loss=loss)
        optimizer.step()  # Always AFTER capture.

snapshot = run.snapshots[0]
print(snapshot.data["loss"], snapshot.data["parameters"])
run.playback(backend="plotly").save("training.html")

Training · Multi-Run Comparison

Training Run Comparison

Compare entire training run histories and experiment trajectories across multiple hyperparameters or optimizers. Aligns training steps across runs, evaluates loss convergence differences, calculates final metric deltas, and flags missing metrics.

Generated by SeeDL 0.5.0 via sd.compare_runs · Compares loss and accuracy trajectories between Baseline SGD and AdamW on a small synthetic CPU demonstration (relative performance varies by task)
View the Python source
import torch
from torch import nn
import seedl as sd

# Small deterministic synthetic demonstration (32 samples, 4 features)
# Relative performance varies with architecture, hyperparameters, and dataset.
torch.manual_seed(42)
X = torch.randn(32, 4)
y = ((X[:, 0] * 1.5 - X[:, 1] * 0.8 + X[:, 2] * 0.5) > 0.0).long()

def make_model():
    torch.manual_seed(100)
    return nn.Sequential(nn.Linear(4, 8), nn.ReLU(), nn.Linear(8, 2))

model_sgd, model_adamw = make_model(), make_model()
model_adamw.load_state_dict(model_sgd.state_dict())
criterion = nn.CrossEntropyLoss()

opt_sgd = torch.optim.SGD(model_sgd.parameters(), lr=0.1)
opt_adamw = torch.optim.AdamW(model_adamw.parameters(), lr=0.05, weight_decay=1e-4)

run_sgd, run_adamw = {"loss": [], "accuracy": []}, {"loss": [], "accuracy": []}
for _ in range(4):
    opt_sgd.zero_grad()
    l_sgd = criterion(model_sgd(X), y)
    l_sgd.backward()
    opt_sgd.step()
    with torch.no_grad():
        run_sgd["loss"].append(round(float(l_sgd.item()), 4))
        run_sgd["accuracy"].append(round(float((model_sgd(X).argmax(-1) == y).float().mean().item()), 4))

    opt_adamw.zero_grad()
    l_adamw = criterion(model_adamw(X), y)
    l_adamw.backward()
    opt_adamw.step()
    with torch.no_grad():
        run_adamw["loss"].append(round(float(l_adamw.item()), 4))
        run_adamw["accuracy"].append(round(float((model_adamw(X).argmax(-1) == y).float().mean().item()), 4))

view = sd.compare_runs(run_sgd, run_adamw, name_a="Baseline SGD", name_b="AdamW Optimizer")
view.save("runs_compare.html")
print(view.data["metrics"])

Prepare a Bell pair

1. Start at |00⟩
Both qubits begin in zero.

2. Apply H to q₀
Creates (|00⟩ + |01⟩)/√2.

3. Apply CX(q₀, q₁)
Flips q₁ when q₀ = 1.

Final state

|Φ⁺⟩ = (|00⟩ + |11⟩)/√2

Measurement gives 00 or 11 with 50% probability each. Each qubit's reduced state is I/2, with entropy 1 bit.
